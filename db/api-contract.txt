function api.fn_allocate_course(p_course_id integer, p_bed_ids integer[], p_from date, p_to date)
  -- Allocate a course into the given beds from p_from (first night) to p_to (departure day). One allocation per bed, status ALLOCATED, each given the course's next free individual (no upper limit, D31). Clashes warn unless their rule is BLOCK, which refuses the whole request (D30). Nights outside the course dates warn (0010). Audited. Returns each new allocation with its warnings.
  returns TABLE(allocation_id integer, bed_id integer, warnings text[])

function api.fn_bed_bars(p_from date, p_to date)
  -- Bars for every bed that overlap p_from to p_to (exclusive): kind allocation, external or outage, with their full dates (end exclusive). on_outage flags a booking that falls on an out-of-action bed.
  returns TABLE(bed_id integer, kind text, id integer, start_date date, end_date date, course_id integer, course_code text, individual_no integer, external_ref text, external_name text, status text, set_on_level text, reason text, on_outage boolean)

function api.fn_cancel_allocation(p_allocation_id integer)
  -- Cancel one allocation (kept, with status CANCELLED, so it stays in the audit history). Audited.
  returns TABLE(allocation_id integer, status text)

function api.fn_cancel_course(p_course_id integer)
  -- Cancel a course and every live allocation on it (all kept, as CANCELLED). Each cancellation is audited. Returns how many allocations were cancelled.
  returns TABLE(course_id integer, allocations_cancelled integer)

function api.fn_candidate_beds(p_course_id integer, p_from date, p_to date, p_beds integer DEFAULT NULL::integer)
  -- Beds a course could be booked into from p_from (first night) to p_to (departure day): every bed on the course's site (all beds when it has none), in estate order. free: no booking or outage on those nights. warnings: the clashes the active rules would raise; blocks: one of them is BLOCK, so fn_allocate_course would refuse. course_in_room: the course already has a bed in that room on those nights. Given p_beds, that many free beds are suggested (suggested, rank), keeping the course together and filling rooms (D34); fewer when not enough are free. Book them with fn_allocate_course.
  returns TABLE(bed_id integer, bed_name text, room_id integer, room_name text, floor_name text, block_name text, building_name text, site_name text, free boolean, course_in_room boolean, warnings text[], blocks boolean, rank integer, suggested boolean)

function api.fn_check_allocation(p_bed_ids integer[], p_from date, p_to date, p_course_id integer DEFAULT NULL::integer)
  -- Clashes for allocating the given beds from p_from to p_to (exclusive), before anything is saved: existing bookings and outages on those beds. blocks is true when the clash's rule is BLOCK, so saving would be refused (D30); otherwise it saves with a warning. Given p_course_id, also a COURSE_DATES row (bed_id null) when the nights go outside that course's dates (0010).
  returns TABLE(bed_id integer, kind text, ref text, start_date date, end_date date, message text, blocks boolean)

function api.fn_create_course(p_code text, p_name text, p_start_date date, p_end_date date, p_scheduled integer DEFAULT 0, p_loaded integer DEFAULT 0, p_site_id integer DEFAULT NULL::integer, p_status text DEFAULT 'REQUESTED'::text, p_notes text DEFAULT NULL::text)
  -- Create a course: code (unique in the tenant, case-insensitive), name, start_date (first night) and end_date (departure day), scheduled and loaded headcounts (loaded_at is set when loaded is above 0), an optional site, status REQUESTED (default) or CONFIRMED, and notes. Audited. Returns the new course_id.
  returns TABLE(course_id integer, code text)

function api.fn_move_allocation(p_allocation_id integer, p_bed_id integer DEFAULT NULL::integer, p_from date DEFAULT NULL::date, p_to date DEFAULT NULL::date)
  -- Move an allocation to another bed and/or other dates (any argument left null stays as it is). Clashes warn unless their rule is BLOCK (D30); nights outside the course dates warn (0010). Audited. Returns the allocation with its warnings.
  returns TABLE(allocation_id integer, bed_id integer, start_date date, end_date date, warnings text[])

function api.fn_set_status(p_allocation_id integer, p_status text)
  -- Set an allocation's status: ALLOCATED, BOOKED, CHECKED_IN, CHECKED_OUT, NO_SHOW or CANCELLED. Any change is allowed for now (transitions deferred, D31). Sets or clears the check-in and check-out times to match. Audited when the status changes.
  returns TABLE(allocation_id integer, status text, checked_in_at timestamp without time zone, checked_out_at timestamp without time zone)

function api.fn_update_course(p_course_id integer, p_code text DEFAULT NULL::text, p_name text DEFAULT NULL::text, p_start_date date DEFAULT NULL::date, p_end_date date DEFAULT NULL::date, p_scheduled integer DEFAULT NULL::integer, p_site_id integer DEFAULT NULL::integer, p_clear_site boolean DEFAULT false, p_status text DEFAULT NULL::text, p_notes text DEFAULT NULL::text)
  -- Change a course: code, name, start_date, end_date, scheduled, site (p_clear_site removes it), status (REQUESTED or CONFIRMED; a cancelled course can be reinstated this way, its bookings stay cancelled) and notes (empty clears them). Anything left null stays as it is. Audited. Returns warnings for live bookings the change leaves outside the course dates or on another site.
  returns TABLE(course_id integer, warnings text[])

function api.fn_update_loaded(p_course_id integer, p_loaded integer)
  -- Set a course's loaded (registered) headcount and stamp loaded_at with now. Audited. Returns the new figure and whether it is above scheduled.
  returns TABLE(course_id integer, loaded integer, loaded_at timestamp without time zone, loaded_above_scheduled boolean)

function api.fn_week_summary(p_from date, p_weeks integer)
  -- For each location above bed level (key as in api.locations) and each of p_weeks weeks from p_from: the busiest night (fewest free beds, earliest on a tie) and its beds by category: free, out_of_action, external, or course (one row per course_id). Categories with no beds are left out. The UI bands courses by colour from api.course_figures (D29).
  returns TABLE(key text, week_start date, busiest_night date, category text, course_id integer, beds bigint)

function api.fn_weekly_overview(p_from date, p_weeks integer, p_site_ids integer[] DEFAULT NULL::integer[])
  -- One row per week from p_from (p_weeks weeks), for beds in the given sites (all when p_site_ids is null). beds_available: beds not out of action on the week's busiest night (fewest free beds, earliest on a tie), split into allocated (course bookings), external and free. scheduled and loaded: totals for courses running on at least one night of the week.
  returns TABLE(week_start date, busiest_night date, beds_available bigint, allocated bigint, external bigint, free bigint, scheduled bigint, loaded bigint, courses bigint)

function api.reset_demo_data()

view api.course_figures
  -- Per course: scheduled (planned), loaded (registered, as of loaded_at) and allocated (individuals with a bed, not cancelled). pct_scheduled and pct_loaded are allocated as a percentage of each, to 1 decimal place, null when that figure is 0. Colour bands are applied by the UI from the user's thresholds (D29). site_id is the course's site (0011).
  course_id integer
  code text
  name text
  start_date date
  end_date date
  status text
  scheduled integer
  loaded integer
  loaded_at timestamp without time zone
  allocated bigint
  loaded_above_scheduled boolean
  pct_scheduled numeric
  pct_loaded numeric
  site_id integer

  -- Health check: status, time, the selected tenant and its bed count.
  status text
  checked_at timestamp with time zone
  tenant character varying(30)
  bed_count bigint

view api.locations
  -- The estate as a tree: one row per site, building, block, floor, room and bed. key is unique (e.g. "room:12"), parent_key points to the row above, sort_order orders siblings, bed_count is the number of beds at or below.
  key text
  parent_key text
  level text
  id integer
  name text
  sort_order integer
  bed_count bigint

  -- Tenants that can be selected with the X-Tenant header.
  code character varying(30)
  name character varying(100)
  is_demo character(1)

view api.warnings
  -- Warnings for the warnings strip, whatever the colour basis: courses with loaded above scheduled (D14), bookings on out-of-action beds, double bookings (one row per booking involved), and course bookings with nights outside their course's dates (strict, 0010). Clash severities come from their rules (D30). site_id is the bed's site, or the course's for course-level warnings (0011).
  kind text
  severity text
  message text
  course_id integer
  allocation_id integer
  bed_id integer
  start_date date
  end_date date
  site_id integer
