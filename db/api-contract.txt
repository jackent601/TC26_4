function api.fn_bed_bars(p_from date, p_to date)
  -- Bars for every bed between p_from and p_to (exclusive): kind allocation, external or outage, with dates clipped to the range. on_outage flags a booking that falls on an out-of-action bed.
  returns TABLE(bed_id integer, kind text, id integer, start_date date, end_date date, course_id integer, course_code text, individual_no integer, external_ref text, external_name text, status text, set_on_level text, reason text, on_outage boolean)

function api.fn_check_allocation(p_bed_ids integer[], p_from date, p_to date)
  -- Clashes for allocating the given beds from p_from to p_to (exclusive), before anything is saved: existing bookings and outages on those beds. Clashes warn by default (D11).
  returns TABLE(bed_id integer, kind text, ref text, start_date date, end_date date, message text)

function api.fn_week_summary(p_from date, p_weeks integer)
  -- For each location above bed level (key as in api.locations) and each of p_weeks weeks from p_from: the busiest night (fewest free beds, earliest on a tie) and its beds by category: free, out_of_action, external, or course (one row per course_id). Categories with no beds are left out. The UI bands courses by colour from api.course_figures (D29).
  returns TABLE(key text, week_start date, busiest_night date, category text, course_id integer, beds bigint)

function api.reset_demo_data()
  -- Wipe and reload the current tenant's reference and sample data. Demo tenants only.
  returns TABLE(tenant text, day0 date, beds bigint, courses bigint, allocations bigint, outages bigint)

function api.use_tenant(p_code text, p_local boolean DEFAULT true)
  -- Select a tenant for the current transaction. Normally done for you from the X-Tenant header.
  returns text

view api.course_figures
  -- Per course: scheduled (planned), loaded (registered, as of loaded_at) and allocated (individuals with a bed, not cancelled). pct_scheduled and pct_loaded are allocated as a percentage of each, to 1 decimal place, null when that figure is 0. Colour bands are applied by the UI from the user's thresholds (D29).
  course_id integer
  code text
  name text
  start_date date
  end_date date
  status text
  scheduled integer
  loaded integer
  loaded_at timestamp without time zone
  allocated bigint
  loaded_above_scheduled boolean
  pct_scheduled numeric
  pct_loaded numeric

view api.health
  -- Health check: status, time, the selected tenant and its bed count.
  status text
  checked_at timestamp with time zone
  tenant character varying(30)
  bed_count bigint

view api.locations
  -- The estate as a tree: one row per site, building, block, floor, room and bed. key is unique (e.g. "room:12"), parent_key points to the row above, sort_order orders siblings, bed_count is the number of beds at or below.
  key text
  parent_key text
  level text
  id integer
  name text
  sort_order integer
  bed_count bigint

view api.tenants
  -- Tenants that can be selected with the X-Tenant header.
  code character varying(30)
  name character varying(100)
  is_demo character(1)

view api.warnings
  -- Warnings for the warnings strip, whatever the colour basis: courses with loaded above scheduled (D14), bookings on out-of-action beds, and double bookings (one row per booking involved).
  kind text
  severity text
  message text
  course_id integer
  allocation_id integer
  bed_id integer
  start_date date
  end_date date
